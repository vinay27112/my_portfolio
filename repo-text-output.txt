import { useEffect } from "react";
import Lenis from "lenis";
import gsap from "gsap";
import { ScrollTrigger } from "gsap/ScrollTrigger";

gsap.registerPlugin(ScrollTrigger);

export default function useLenis() {
  useEffect(() => {
    const reduce = window.matchMedia(
      "(prefers-reduced-motion: reduce)",
    ).matches;
    if (reduce) return;

    const lenis = new Lenis({
      duration: 1.2,
      easing: (t) => 1 - Math.pow(1 - t, 3), // easeOutCubic — smooth, no slow start
      smoothWheel: true,
    });

    // Keep ScrollTrigger's notion of scroll position in sync with Lenis's
    // virtual (transform-based) scroll — without this, scroll-triggered
    // reveals can fire at the wrong point and feel glitchy.
    lenis.on("scroll", ScrollTrigger.update);

    // GSAP's ticker reports time in SECONDS; Lenis expects MILLISECONDS.
    // Feeding it raw seconds made Lenis think almost no time had passed
    // each frame, which is what caused the slow/stuttery scroll.
    function raf(time) {
      lenis.raf(time * 1000);
    }
    gsap.ticker.add(raf);
    gsap.ticker.lagSmoothing(0);

    return () => {
      gsap.ticker.remove(raf);
      lenis.destroy();
    };
  }, []);
}

import { useEffect } from "react";
import gsap from "gsap";
import { ScrollTrigger } from "gsap/ScrollTrigger";

gsap.registerPlugin(ScrollTrigger);

// Animates every [data-reveal] element inside containerRef as it scrolls
// into view. Elements with [data-reveal-group] get a staggered entrance
// instead of animating one by one.
export default function useScrollReveal(containerRef, deps = []) {
  useEffect(() => {
    const root = containerRef.current;
    if (!root) return;

    const reduce = window.matchMedia(
      "(prefers-reduced-motion: reduce)",
    ).matches;

    const ctx = gsap.context(() => {
      const groups = root.querySelectorAll("[data-reveal-group]");
      groups.forEach((group) => {
        const items = group.querySelectorAll("[data-reveal]");
        if (!items.length) return;
        if (reduce) {
          gsap.set(items, { opacity: 1, y: 0 });
          return;
        }
        gsap.set(items, { opacity: 0, y: 24 });
        ScrollTrigger.batch(items, {
          start: "top 88%",
          onEnter: (batch) =>
            gsap.to(batch, {
              opacity: 1,
              y: 0,
              duration: 0.6,
              stagger: 0.08,
              ease: "power2.out",
            }),
          once: true,
        });
      });

      const singles = root.querySelectorAll(
        "[data-reveal]:not([data-reveal-group] [data-reveal])",
      );
      singles.forEach((el) => {
        if (reduce) {
          gsap.set(el, { opacity: 1, y: 0 });
          return;
        }
        gsap.set(el, { opacity: 0, y: 24 });
        gsap.to(el, {
          opacity: 1,
          y: 0,
          duration: 0.7,
          ease: "power2.out",
          scrollTrigger: {
            trigger: el,
            start: "top 90%",
            once: true,
          },
        });
      });
    }, root);

    return () => ctx.revert();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, deps);
}

import { useEffect, useRef } from "react";

import gsap from "gsap";
import { profile, projects } from "../data";
import ProjectCard from "../components/ProjectCard.jsx";
import MoonHero from "../components/MoonHero.jsx";
import useScrollReveal from "../lib/useScrollReveal.js";

export default function Home() {
  const heroRef = useRef(null);
  const pageRef = useRef(null);
  useScrollReveal(pageRef, []);

  useEffect(() => {
    const reduce = window.matchMedia(
      "(prefers-reduced-motion: reduce)",
    ).matches;
    const root = heroRef.current;
    if (!root) return;

    const eyebrow = root.querySelector("[data-hero-eyebrow]");
    const heading = root.querySelector("[data-hero-heading]");
    const tagline = root.querySelector("[data-hero-tagline]");
    const cta = root.querySelector("[data-hero-cta]");
    const art = root.querySelector("[data-hero-art]");

    if (reduce) {
      gsap.set([eyebrow, heading, tagline, cta, art], { opacity: 1, y: 0 });
      return;
    }

    gsap.set([eyebrow, heading, tagline, cta], { opacity: 0, y: 18 });
    gsap.set(art, { opacity: 0, scale: 0.96 });

    const tl = gsap.timeline({ defaults: { ease: "power2.out" } });
    tl.to(art, { opacity: 1, scale: 1, duration: 0.9 }, 0)
      .to(eyebrow, { opacity: 1, y: 0, duration: 0.5 }, 0.15)
      .to(heading, { opacity: 1, y: 0, duration: 0.6 }, 0.28)
      .to(tagline, { opacity: 1, y: 0, duration: 0.6 }, 0.4)
      .to(cta, { opacity: 1, y: 0, duration: 0.5 }, 0.52);
  }, []);

  const featured = projects.filter((p) => p.status === "live").slice(0, 2);

  return (
    <div ref={pageRef}>
      <section ref={heroRef} className="relative overflow-hidden">
        <div data-hero-art className="absolute inset-0 -z-10">

          <p data-hero-eyebrow className="text-sm font-medium text-ink-soft mb-4">
            {profile.location} · Available for full stack & Web3 roles
          </p>
          <h1
            data-hero-heading
            className="text-4xl sm:text-5xl font-semibold text-ink tracking-tight mb-5 max-w-xl"
          >
            {profile.name}
          </h1>
          <p data-hero-tagline className="text-lg text-ink-soft max-w-lg mb-8">
            {profile.tagline}
          </p>
          <div data-hero-cta className="flex flex-wrap gap-3">
            <Link
              to="/projects"
              className="px-5 py-2.5 rounded-full bg-ink text-bg text-sm font-medium hover:bg-ink-soft transition"
            >
              View projects
            </Link>
            <Link
              to="/contact"
              className="px-5 py-2.5 rounded-full border border-line text-ink-soft text-sm font-medium hover:text-ink hover:border-ink-dim transition"
            >
              Get in touch
            </Link>
          </div>
        </div>
      </section>

      <section className="max-w-5xl mx-auto px-6 pb-8">
